import random
from pathlib import Path

README = Path("README.md")
QUOTES = Path("data/quotes.txt")
STATE = Path("data/.quote_state")

quotes = [
    q.strip()
    for q in QUOTES.read_text(encoding="utf-8").split("\n\n")
    if q.strip()
]

# Load previous order or create a new shuffled order
if STATE.exists():
    order = STATE.read_text().splitlines()
else:
    order = []

# If we've used every quote, reshuffle
if not order:
    indices = list(range(len(quotes)))
    random.shuffle(indices)
    order = [str(i) for i in indices]

# Pick next quote
index = int(order.pop(0))
quote = quotes[index]

# Save remaining order
STATE.write_text("\n".join(order), encoding="utf-8")

replacement = f"""<!--QUOTE_START-->

> 💭 **Today's Repository Thought**
>
> *{quote}*

<!--QUOTE_END-->"""

content = README.read_text(encoding="utf-8")

start = content.index("<!--QUOTE_START-->")
end = content.index("<!--QUOTE_END-->") + len("<!--QUOTE_END-->")

content = content[:start] + replacement + content[end:]

README.write_text(content, encoding="utf-8")
